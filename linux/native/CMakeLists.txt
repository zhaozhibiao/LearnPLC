cmake_minimum_required(VERSION 3.20)
project(st_workbench_linux LANGUAGES C CXX)
if(NOT UNIX)
  message(FATAL_ERROR "Build Linux production services on their Linux target")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.2.0 REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(SQLite3 REQUIRED)
find_path(ARCHIVE_INCLUDE_DIR archive.h HINTS "${OPENSSL_INCLUDE_DIR}" REQUIRED)
find_library(ARCHIVE_LIBRARY NAMES archive libarchive.so.13 REQUIRED)
if(EXISTS "${OPENSSL_INCLUDE_DIR}/${CMAKE_LIBRARY_ARCHITECTURE}/openssl/opensslconf.h")
  include_directories(SYSTEM "${OPENSSL_INCLUDE_DIR}/${CMAKE_LIBRARY_ARCHITECTURE}")
endif()
add_library(st-native-common src/common.cpp)
target_include_directories(st-native-common PUBLIC include)
target_link_libraries(st-native-common PUBLIC nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(st-native-common PRIVATE -Wall -Wextra -Wpedantic)
function(st_program name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE st-native-common)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
endfunction()
